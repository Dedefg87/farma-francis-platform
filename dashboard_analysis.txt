import { router, protectedProcedure } from './_core/trpc';

/**
 * Router para drill-down de dados do dashboard
 * Permite visualizar detalhes de métricas agregadas
 */

export const dashboardDrilldownRouter = router({
  /**
   * Buscar conversas de uma hora específica
   * Usado no drill-down do gráfico "Contatos por hora"
   */
  getConversationsByHour: protectedProcedure
    .input(
      z.object({
        hour: z.number().min(0).max(23), // Hora do dia (0-23)
        date: z.string().optional(), // Data no formato YYYY-MM-DD (padrão: hoje)
        status: z.enum(['all', 'chats', 'abandoned']).optional().default('all'),
      })
    )
    .query(async ({ input }) => {
      const db = await getDb();
      if (!db) throw new Error('Database not available');

      // Calcular range de tempo para a hora específica
      const targetDate = input.date ? new Date(input.date) : new Date();
      targetDate.setHours(input.hour, 0, 0, 0);
      
      const startTime = new Date(targetDate);
      const endTime = new Date(targetDate);
      endTime.setHours(input.hour + 1, 0, 0, 0);

      // Construir filtro de status
      let statusFilter;
      if (input.status === 'chats') {
        statusFilter = sql`${conversations.status} IN ('assigned', 'closed')`;
      } else if (input.status === 'abandoned') {
        statusFilter = sql`${conversations.status} = 'abandoned'`;
      } else {
        statusFilter = sql`1=1`; // Todos
      }

      // Buscar conversas da hora específica
      const result = await db
        .select({
          id: conversations.id,
          customerName: conversations.customerName,
          customerPhone: conversations.customerPhone,
          channel: conversations.channel,
          status: conversations.status,
          openedAt: conversations.openedAt,
          closedAt: conversations.closedAt,
          assignedTo: conversations.assignedTo,
          queueId: conversations.queueId,
        })
        .from(conversations)
        .where(
          and(
            gte(conversations.openedAt, startTime),
            lte(conversations.openedAt, endTime),
            statusFilter
          )
        )
        .orderBy(conversations.openedAt);

      return {
        hour: input.hour,
        date: targetDate.toISOString().split('T')[0],
        startTime: startTime.toISOString(),
        endTime: endTime.toISOString(),
        total: result.length,
        conversations: result,
      };
    }),

  /**
   * Buscar conversas de um período específico
   * Usado para drill-down de métricas agregadas
   */
  getConversationsByPeriod: protectedProcedure
    .input(
      z.object({
        startDate: z.string(), // YYYY-MM-DD
        endDate: z.string(), // YYYY-MM-DD
        status: z.enum(['all', 'open', 'assigned', 'closed']).optional().default('all'),
        channel: z.enum(['all', 'whatsapp', 'instagram', 'telegram', 'email']).optional().default('all'),
        limit: z.number().min(1).max(100).optional().default(50),
        offset: z.number().min(0).optional().default(0),
      })
    )
    .query(async ({ input }) => {
      const db = await getDb();
      if (!db) throw new Error('Database not available');

      const startTime = new Date(input.startDate);
      startTime.setHours(0, 0, 0, 0);
      
      const endTime = new Date(input.endDate);
      endTime.setHours(23, 59, 59, 999);

      // Construir filtros
      const filters = [
        gte(conversations.openedAt, startTime),
        lte(conversations.openedAt, endTime),
      ];

      if (input.status !== 'all') {
        filters.push(sql`${conversations.status} = ${sql.raw(`'${input.status}'`)}`);
      }

      if (input.channel !== 'all') {
        filters.push(sql`${conversations.channel} = ${sql.raw(`'${input.channel}'`)}`);
      }

      // Buscar conversas
      const result = await db
        .select({
          id: conversations.id,
          customerName: conversations.customerName,
          customerPhone: conversations.customerPhone,
          channel: conversations.channel,
          status: conversations.status,
          openedAt: conversations.openedAt,
          closedAt: conversations.closedAt,
          assignedTo: conversations.assignedTo,
          queueId: conversations.queueId,
        })
        .from(conversations)
        .where(and(...filters))
        .orderBy(conversations.openedAt)
        .limit(input.limit)
        .offset(input.offset);

      // Contar total (para paginação)
      const [{ total }] = await db
        .select({ total: sql<number>`count(*)` })
        .from(conversations)
        .where(and(...filters));

      return {
        startDate: input.startDate,
        endDate: input.endDate,
        total,
        showing: result.length,
        offset: input.offset,
        limit: input.limit,
        hasMore: input.offset + result.length < total,
        conversations: result,
      };
    }),
});

// Utilitário para formatar segundos em HH:MM:SS
function formatSecondsToHMS(totalSeconds: number): string {
  const hours = Math.floor(totalSeconds / 3600);
  const minutes = Math.floor((totalSeconds % 3600) / 60);
  const seconds = Math.round(totalSeconds % 60);
  return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
}

import { z } from "zod";
import { protectedProcedure, router } from "./_core/trpc";
import { getDb } from "./db";
import { tickets, customers, receivedMessages, messages, employees } from "../drizzle/schema";
import { eq, and, gte, lte, sql, desc } from "drizzle-orm";

/**
 * Dashboard router for real-time statistics
 */
export const dashboardRouter = router({
  /**
   * Get main dashboard statistics
   */
  getStats: protectedProcedure
    .input(
      z.object({
        startDate: z.string().optional(),
        endDate: z.string().optional(),
      })
    )
    .query(async ({ input }) => {
      const db = await getDb();
      if (!db)
        return {
          totalTickets: 0,
          uniqueCustomers: 0,
          messagesSent: 0,
          messagesReceived: 0,
          avgHandlingTime: 0,
        };

      // Count tickets
      const ticketCount = await db
        .select({ count: sql<number>`COUNT(*)` })
        .from(tickets);

      // Count unique customers
      const customerCount = await db
        .select({ count: sql<number>`COUNT(DISTINCT ${customers.id})` })
        .from(customers);

      // Count messages sent
      const sentCount = await db
        .select({ count: sql<number>`COUNT(*)` })
        .from(messages)
        .where(eq(messages.direction, "outbound"));

      // Count messages received
      const receivedCount = await db
        .select({ count: sql<number>`COUNT(*)` })
        .from(receivedMessages);

      // Calculate average handling time (in minutes)
      const avgTime = await db
        .select({ avg: sql<number>`AVG(${tickets.timeSpentMinutes})` })
        .from(tickets)
        .where(sql`${tickets.timeSpentMinutes} > 0`);

      return {
        totalTickets: Number(ticketCount[0]?.count || 0),
        uniqueCustomers: Number(customerCount[0]?.count || 0),
        messagesSent: Number(sentCount[0]?.count || 0),
        messagesReceived: Number(receivedCount[0]?.count || 0),
        avgHandlingTime: Math.round(Number(avgTime[0]?.avg || 0)),
      };
    }),

  /**
   * Get messages by hour for chart
   */
  getMessagesByHour: protectedProcedure.query(async () => {
    const db = await getDb();
    if (!db) return [];

    // Get messages grouped by hour
    const result = await db
      .select({
        hour: sql<number>`HOUR(FROM_UNIXTIME(${receivedMessages.timestamp} / 1000))`,
        count: sql<number>`COUNT(*)`,
      })
      .from(receivedMessages)
      .groupBy(sql`HOUR(FROM_UNIXTIME(${receivedMessages.timestamp} / 1000))`)
      .orderBy(sql`HOUR(FROM_UNIXTIME(${receivedMessages.timestamp} / 1000))`);

    return result.map((r) => ({
      hour: Number(r.hour),
      count: Number(r.count),
    }));
  }),

  /**
   * Get tickets by status
   */
  getTicketsByStatus: protectedProcedure.query(async () => {
    const db = await getDb();
    if (!db) return [];

    const result = await db
      .select({
        status: tickets.status,
        count: sql<number>`COUNT(*)`,
      })
      .from(tickets)
      .groupBy(tickets.status);

    return result.map((r) => ({
      status: r.status,
      count: Number(r.count),
    }));
  }),

  /**
   * Get tickets by channel
   */
  getTicketsByChannel: protectedProcedure.query(async () => {
    const db = await getDb();
    if (!db) return [];

    const result = await db
      .select({
        channel: tickets.channel,
        count: sql<number>`COUNT(*)`,
      })
      .from(tickets)
      .where(sql`${tickets.channel} IS NOT NULL`)
      .groupBy(tickets.channel);

    return result.map((r) => ({
      channel: r.channel,
      count: Number(r.count),
    }));
  }),

  /**
   * Get employee performance metrics
   */
  getEmployeeMetrics: protectedProcedure.query(async () => {
    const db = await getDb();
    if (!db) return [];

    // Get tickets assigned to each employee
    const result = await db
      .select({
        employeeId: tickets.assignedTo,
        ticketsHandled: sql<number>`COUNT(*)`,
        avgTime: sql<number>`AVG(${tickets.timeSpentMinutes})`,
      })
      .from(tickets)
      .where(sql`${tickets.assignedTo} IS NOT NULL`)
      .groupBy(tickets.assignedTo);

    return result.map((r) => ({
      employeeId: r.employeeId,
      ticketsHandled: Number(r.ticketsHandled),
      avgHandlingTime: Math.round(Number(r.avgTime || 0)),
    }));
  }),

  /**
   * Get recent tickets
   */
  getRecentTickets: protectedProcedure
    .input(
      z.object({
        limit: z.number().optional().default(10),
      })
    )
    .query(async ({ input }) => {
      const db = await getDb();
      if (!db) return [];

      const result = await db
        .select()
        .from(tickets)
        .orderBy(desc(tickets.createdAt))
        .limit(input.limit);

      return result;
    }),
});
